\begin{frame}{1. Questão Central e a Ruptura Quântica (Planck-Einstein)}
  \vspace{-0.15cm}
  \begin{block}{Tema 2 — Equipe: Logan e Samuel Farias (TI0060 - Prof. Joacir Andrade)}
    \textbf{Questão Central:} Como a \textbf{frequência ($f$)} e a \textbf{intensidade ($I$)} da luz incidente influenciam a emissão de elétrons de uma superfície metálica, e como a simulação verifica as previsões da teoria quântica?
  \end{block}

  \vspace{0.1cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Falha da Teoria Clássica Ondulatória}
        \footnotesize
        \begin{itemize}
          \item A energia da onda eletromagnética dependeria apenas da sua amplitude (intensidade $I \propto |\mathbf{E}_0|^2$).
          \item \textbf{Previsão incorreta 1:} Luz de qualquer frequência deveria arrancar elétrons se a intensidade $I$ fosse alta o suficiente.
          \item \textbf{Previsão incorreta 2:} $K_{\max}$ deveria crescer com o aumento da intensidade luminosa $I$.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Solução Quântica (Planck 1900 / Einstein 1905)}
        \footnotesize
        \begin{itemize}
          \item A luz é constituída por pacotes discretos de energia (\textbf{fótons}):
          \[ E_f = h f = \frac{h c}{\lambda} \]
          \item $h = 6{,}626 \times 10^{-34}\text{ J}\cdot\text{s} \approx 4{,}136 \times 10^{-15}\text{ eV}\cdot\text{s}$.
          \item A absorção ocorre individualmente (\textbf{1 fóton $\rightarrow$ 1 elétron}).
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 2: MODELAGEM FÍSICA, FUNÇÃO TRABALHO E EQUAÇÃO DE EINSTEIN
% =============================================================================
\begin{frame}{2. Modelagem Física: Função Trabalho ($\phi$) e Equação de Einstein}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \footnotesize
      \textbf{Balanço de Energia Quântico no Metal:}
      \begin{itemize}
        \item \textbf{Função Trabalho ($\phi \equiv W_0$):} Energia mínima para extrair um elétron do nível de Fermi até o vácuo externo.
        \item \textbf{Frequência de Corte ($f_0$):} Limiar abaixo do qual \textbf{não há emissão} ($hf < \phi$):
        \[ f_0 = \frac{\phi}{h}, \qquad \lambda_0 = \frac{c}{f_0} = \frac{hc}{\phi} \]
        \item \textbf{Equação de Einstein ($f \ge f_0$):}
        \[ K_{\max} = E_c = h f - \phi = h(f - f_0) \]
        \item \textbf{Potencial de Corte ($V_0$):} Tensão reversa que freia os fotoelétrons mais rápidos ($q V_0 = K_{\max}$):
        \[ V_0 = \frac{h}{q} f - \frac{\phi}{q}, \qquad v_{\max} = \sqrt{\frac{2 K_{\max}}{m_e}} \]
      \end{itemize}
    \end{column}

\begin{frame}{4. Simulação 2: Intensidade ($I$) vs. Frequência ($f$) na Curva $I_{\text{foto}} \times V$}
  \centering
  \includegraphics[width=0.92\textwidth]{fig2_corrente_vs_tensao.png}
  \vspace{-0.15cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \scriptsize
      \textbf{(a) Variação da Intensidade ($f$ fixa):} Aumentar $I$ multiplica o número de fótons/s $\Rightarrow$ a fotocorrente de saturação cresce proporcionalmente, mas o potencial de corte $-V_0 = -0{,}70\text{ V}$ (e $K_{\max}$) \textbf{não se altera}.
    \end{column}
    \begin{column}{0.48\textwidth}
      \scriptsize
      \textbf{(b) Variação da Frequência (Taxa de fótons fixa):} Aumentar $f$ eleva a energia individual de cada fóton ($hf$) $\Rightarrow$ $K_{\max}$ cresce e o potencial de corte $-V_0$ torna-se mais negativo.
    \end{column}
  \end{columns}
\end{frame}

  \begin{block}{Síntese Final e Aplicações em Engenharia de Telecomunicações e Optoeletrônica}
