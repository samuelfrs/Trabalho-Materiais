\begin{frame}{1. O Problema: Comportamento da Emissão Fotoelétrica}
  \vspace{-0.15cm}
  \begin{block}{Tema 2 — Equipe: Logan Oliveira e Samuel Farias (TI0060 - Prof. Joacir Andrade)}
    \textbf{Questão Central:} Como a frequência ($f$) e a intensidade ($I$) da radiação governam a emissão de elétrons de uma superfície metálica, e como podemos modelar e verificar esse comportamento computacionalmente?
  \end{block}

  \vspace{0.1cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{O Desafio Físico a Explicar}
        \footnotesize
        \begin{itemize}
          \item \textbf{Frequência de Corte Estrita ($f_0$):} Abaixo de $f_0$, nenhum elétron é liberado do metal, por maior que seja a intensidade luminosa.
          \item \textbf{Independência de $K_{\max}$ frente a $I$:} Aumentar a intensidade aumenta a quantidade de elétrons, mas não altera a energia máxima individual ($K_{\max}$).
          \item \textbf{Resposta Instantânea:} Emissão em $\sim 10^{-9}\text{ s}$, sem atraso de acúmulo contínuo de energia.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Parâmetros e Variáveis do Estudo}
        \footnotesize
        \begin{itemize}
          \item \textbf{Radiação Incidente:} Frequência ($f$), comprimento de onda ($\lambda = c/f$) e intensidade luminosa ($I$).
          \item \textbf{Cátodo Emissor:} Função trabalho ($\phi$) dos 9 metais da Aula 03 (do $\text{Na}$ com $2{,}28\text{ eV}$ à $\text{Pt}$ com $6{,}35\text{ eV}$).
          \item \textbf{Circuito Elétrico:} Potencial de frenagem ($-V_0$) e fotocorrente ($I_{\text{foto}}$).
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 2: COMO RESOLVEMOS: MODELAGEM TEÓRICA E EQUAÇÕES GOVERNANTES
% =============================================================================
\begin{frame}{2. Como Resolvemos: Modelagem Teórica e Equações Governantes}
  \vspace{-0.15cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Quantização da Luz e Barreira do Material}
        \footnotesize
        \textbf{Energia Discreta do Fóton (Planck-Einstein):}
        \[ E_f = h f = \frac{h c}{\lambda} \]
        com $h \approx 4{,}136 \times 10^{-15}\text{ eV}\cdot\text{s} = 6{,}626 \times 10^{-34}\text{ J}\cdot\text{s}$.
        \vspace{0.15cm}

        \textbf{Função Trabalho ($\phi$) e Limiar de Corte ($f_0$):}
        \[ f_0 = \frac{\phi}{h} \qquad \iff \qquad \lambda_0 = \frac{c}{f_0} = \frac{h c}{\phi} \]
        Condição de emissão: $E_f \ge \phi$ (ou $f \ge f_0$, $\lambda \le \lambda_0$).
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{2. Balanço de Einstein e Frenagem no Circuito}
        \footnotesize
        \textbf{Equação de Einstein para o Efeito Fotoelétrico:}
        \[ K_{\max} = h f - \phi = h(f - f_0) = \frac{1}{2} m_e v_{\max}^2 \]
        onde $m_e \approx 9{,}109 \times 10^{-31}\text{ kg}$ é a massa do elétron.
        \vspace{0.15cm}

        \textbf{Potencial de Frenagem ($V_0$) no Circuito:}
        \[ q V_0 = K_{\max} \implies V_0 = \frac{h}{q} f - \frac{\phi}{q} \]
        Reta de inclinação universal $\Delta V_0 / \Delta f = h/q$, permitindo determinar a constante de Planck experimentalmente.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{5. Resultados 2: Intensidade ($I$) vs. Frequência ($f$) na Fotocorrente}
  \centering
  \includegraphics[width=0.92\textwidth]{fig2_corrente_vs_tensao.png}
  \vspace{-0.15cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \scriptsize
      \textbf{(a) Variação da Intensidade ($f$ fixa):} Aumentar $I$ multiplica o número de fótons/s $\Rightarrow$ a fotocorrente de saturação cresce proporcionalmente ($I_{\text{sat}} \propto I$), mas o potencial de corte $-V_0 = -0{,}70\text{ V}$ (e $K_{\max}$) \textbf{permanece rigorosamente fixo}.
    \end{column}
    \begin{column}{0.48\textwidth}
      \scriptsize
      \textbf{(b) Variação da Frequência (Taxa de fótons fixa):} Aumentar $f$ eleva a energia individual de cada fóton ($hf$) $\Rightarrow$ $K_{\max}$ cresce e o potencial de corte $-V_0 = -K_{\max}/q$ torna-se progressivamente mais negativo.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{7. A Simulação Computacional Interativa e Validação ao Vivo}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Simulador Interativo (HTML5 / Canvas)}
        \footnotesize
        \begin{itemize}
          \item Tubo a vácuo com fotocátodo e ânodo com visualização contínua dos fotoelétrons em tempo real.
          \item Seletor rápido dos 9 metais da Aula 03.
          \item Controles de espectro luminoso, intensidade ($0-100\%$) e tensão coletor-cátodo.
          \item \textbf{Botão de Frenagem Rápida:} aplica $-V_0$ e interrompe o fluxo de fotoelétrons imediatamente.
          \item Execução offline em qualquer navegador.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Resposta Definitiva à Questão Central}
        \footnotesize
        \begin{itemize}
          \item \textbf{Frequência ($f$):} Governa a \textbf{qualidade energética} de cada pacote ($E = hf$). Determina SE haverá emissão ($f \ge f_0$), a energia cinética máxima ($K_{\max}$) e o potencial de corte $-V_0$.
          \item \textbf{Intensidade ($I$):} Governa a \textbf{quantidade de pacotes por segundo}. Modula linearmente a fotocorrente de saturação ($I_{\text{sat}}$), sem alterar $K_{\max}$.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
